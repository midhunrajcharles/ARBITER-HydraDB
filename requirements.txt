# Arbiter runtime dependencies.
#
# Note: we deliberately do NOT depend on `hydradb-sdk`. That package targets the
# hosted HydraDB cloud API (context.ingest / client.query). This project targets
# the open-source HydraDB graph engine (github.com/hydra-db/hydradb), which
# speaks OpenCypher over Bolt 5.x and an HTTP JSON API - so a plain HTTP client
# is all that is required.

httpx>=0.27          # HydraDB HTTP query API
fastapi>=0.115       # API layer
uvicorn>=0.30        # ASGI server
pydantic>=2          # request models (pulled in by fastapi, pinned for clarity)
